PASO 1: Hacer Push a Render
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Ejecuta en la terminal:

  git push origin main

Render detectará el cambio y redesplegará automáticamente (5-10 min)


PASO 2: Verificar el Deploy
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Ve a: https://dashboard.render.com
2. Abre el servicio "ecosistema-simulado"
3. Espera a ver: "Deploy live"


PASO 3: Probar el Endpoint
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Abre en tu navegador:

  https://ecosistema-simulado.onrender.com/grafana/

Debe mostrar: "OK"


PASO 4: Ver Datos
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  https://ecosistema-simulado.onrender.com/grafana/data

Debe mostrar: JSON con tus datos de sensores


PASO 5: Configurar Grafana
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Ve a: https://ecosistema-grafana.onrender.com

2. Administration → Plugins → Buscar "JSON API"

3. Si está disponible, instalarlo

4. Connections → Data sources → Add data source

5. Configurar:
   Name: API Ecosistema
   URL: https://ecosistema-simulado.onrender.com/grafana

6. Save & test

7. Crear dashboard con visualizaciones
